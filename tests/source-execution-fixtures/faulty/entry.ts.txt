import type { BrambleContext } from "@fracture/source-target";
import { faultyCheckout } from "./purchase";
import { importedMarker } from "./marker";
export async function checkout(context: BrambleContext): Promise<unknown> {
  await context.store.emit(context.run.id, {lane:"application",kind:"imported_source_executed",title:"Imported checkout",detail:importedMarker(),facts:{marker:importedMarker()}});
  return faultyCheckout(context);
}
